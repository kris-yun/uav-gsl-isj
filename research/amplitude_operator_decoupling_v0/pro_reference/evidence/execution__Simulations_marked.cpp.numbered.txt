0036:     static void weighted_incremental_variance(double value, double weight, double& mean, double& weight_sum, double& weight_squared_sum,
0037:                                               double& variance)
0038:     {
0039:         // Updating Mean and Variance Estimates: An Improved Method D.H.D. West 1979
0040:         weight_sum = weight_sum + weight;
0041:         weight_squared_sum = weight_squared_sum + weight * weight;
0042:         double mean_old = mean;
0043:         mean = mean_old + (weight / weight_sum) * (value - mean_old);
0044:         variance = variance + weight * (value - mean_old) * (value - mean);
0045:     }

0374:     void Simulations::simulateSourceInPosition(const SimulationSource& source, std::vector<float>& hitMap, bool warmup,
0375:                                                int timesteps, float deltaTime, float noiseSTDev) const
0376:     {
0377:         constexpr int numFilamentsIteration = 5;
0378:         size_t max_filaments = settings.maxWarmupIterations * numFilamentsIteration + timesteps * numFilamentsIteration;
0379: 
0380:         // To avoid having to delete filaments from the middle of the vector, which is quite slow, we will ping-pong the active filaments between two vectors
0381:         // at the start of any iteration, one vector (active) will contain all the released filaments and the other one will be empty
0382:         // after each filament has been moved, if it is still active, it will be copied to the other vector
0383:         // then, the active vector changes and the old one is cleared
0384:         std::vector<Filament> filaments1;
0385:         std::vector<Filament> filaments2;
0386:         filaments1.reserve(max_filaments);
0387:         filaments2.reserve(max_filaments);
0388:         std::vector<Filament>* activeFilamentVec = &filaments1;
0389:         std::vector<Filament>* otherFilamentVec = &filaments2;
0390: 
0391:         std::vector<uint16_t> updated(hitMap.size(), 0); // index of the last iteration in which this cell was updated, to avoid double-counting
0392: 
0393:         // warm-up: we don't want to start recording frequency of hits until the shape of the plume has stabilized. Wait until a filament exits the
0394:         // environment through an outlet, or a maximum number of steps
0395:         {
0396:             ZoneScopedN("Warmup");
0397: 
0398:             bool stable = false;
0399:             int iterationCount = 0;
0400:             while (iterationCount < settings.minWarmupIterations || (!stable && iterationCount < settings.maxWarmupIterations))
0401:             {
0402:                 for (size_t i = 0; i < numFilamentsIteration; i++)
0403:                 {
0404:                     activeFilamentVec->emplace_back();
0405:                     activeFilamentVec->back().position = source.getPoint();
0406:                 }
0407: 
0408:                 for (Filament& filament : *activeFilamentVec)
0409:                 {
0410:                     auto indices = measuredHitProb.metadata.coordinatesToIndices(filament.position.x, filament.position.y);
0411: 
0412:                     // move active filaments
0413:                     moveFilament(filament, indices, deltaTime * 2, noiseSTDev);
0414: 
0415:                     // remove filaments
0416:                     if (filamentIsOutside(filament))
0417:                         stable = true;
0418:                     else
0419:                         otherFilamentVec->push_back(filament);
0420:                 }
0421:                 iterationCount++;
0422: 
0423:                 // "other" now contains the list of all the filaments that are still available, so swap the vectors and remove the old list
0424:                 activeFilamentVec->clear();
0425:                 std::swap(activeFilamentVec, otherFilamentVec);
0426:             }
0427:         }
0428: 
0429:         ZoneScopedN("Recording");
0430:         // now, we do the thing
0431:         for (int t = 1; t < timesteps + 1; t++)
0432:         {
0433:             for (int i = 0; i < numFilamentsIteration; i++)
0434:             {
0435:                 activeFilamentVec->emplace_back();
0436:                 activeFilamentVec->back().position = source.getPoint();
0437:             }
0438: 
0439:             for (Filament& filament : *activeFilamentVec)
0440:             {
0441:                 // update map
0442:                 auto indices = measuredHitProb.metadata.coordinatesToIndices(filament.position.x, filament.position.y);
0443:                 size_t index = measuredHitProb.metadata.indexOf(indices);
0444:                 GSL_ASSERT(measuredHitProb.metadata.indicesInBounds(indices));
0445:                 if (markedMultiplicityCounter) (*markedMultiplicityCounter)[index] += 1.0f;
0446:                 // mark as updated so it doesn't count multiple filaments in the same timestep
0447:                 if (updated[index] < t)
0448:                 {
0449:                     hitMap[index]++;
0450:                     updated[index] = t;
0451:                 }
0452: 
0453:                 // move active filaments
0454:                 moveFilament(filament, indices, deltaTime, noiseSTDev);
0455: 
0456:                 // remove filaments
0457:                 if (!filamentIsOutside(filament))
0458:                     otherFilamentVec->push_back(filament);
0459:             }
0460:             activeFilamentVec->clear();
0461:             std::swap(activeFilamentVec, otherFilamentVec);
0462:         }
0463: 
0464:         // convert the total hit count into relative frequency
0465:         for (int i = 0; i < measuredHitProb.data.size(); i++)
0466:         {
0467:             if (measuredHitProb.occupancy[i] == Occupancy::Free)
0468:             {
0469:                 hitMap[i] = hitMap[i] / timesteps;
0470:                 if (markedMultiplicityCounter) (*markedMultiplicityCounter)[i] /= timesteps;
0471:             }
0472:         }
0473:     }

0477:         if (mode == Mode::Point)
0478:             return point;
0479: 
0480:         Vector2 start = metadata.indicesToCoordinates(nqaNode->origin.x, nqaNode->origin.y, false);
0481:         Vector2 end = metadata.indicesToCoordinates(nqaNode->origin.x + nqaNode->size.x, nqaNode->origin.y + nqaNode->size.y, false);
0482: 
0483:         Vector2 randP(Utils::uniformRandomF(start.x, end.x), Utils::uniformRandomF(start.y, end.y));
0484:         GSL_ASSERT(randP.x >= start.x && randP.x < end.x && randP.y >= start.y && randP.y < end.y);
0485:         return randP;

0488:     bool Simulations::moveAlongPath(Vector2& currentPosition, const Vector2& end) const
0489:     {
0490:         Vector2Int indexEnd = measuredHitProb.metadata.coordinatesToIndices(end.x, end.y);
0491:         Vector2Int indexOrigin = measuredHitProb.metadata.coordinatesToIndices(currentPosition.x, currentPosition.y);
0492: 
0493:         if (!measuredHitProb.freeAt(indexOrigin.x, indexOrigin.y))
0494:         {
0495:             return false;
0496:         }
0497: 
0498:         // try to avoid doing the raycast by looking at the pre-computed visibilityMap
0499:         if (indexOrigin == indexEnd || (measuredHitProb.metadata.indicesInBounds(indexEnd) && measuredHitProb.freeAt(indexEnd.x, indexEnd.y) &&
0500:                                         visibilityMap->isVisible(indexOrigin, indexEnd) == Visibility::Visible))
0501:         {
0502:             currentPosition = end;
0503:             return true;
0504:         }
0505: 
0506: #define USE_DDA 0
0507: #if USE_DDA
0508:         Vector2 movement = end - currentPosition;
0509:         DDA::_2D::RayCastInfo raycastInfo =
0510:             DDA::_2D::castRay<GSL::Occupancy>(currentPosition, movement, vmath::length(movement),
0511:                                               DDA::_2D::Map<GSL::Occupancy>(measuredHitProb.occupancy, measuredHitProb.metadata.origin,
0512:                                                                             measuredHitProb.metadata.cellSize, measuredHitProb.metadata.dimensions),
0513:                                               [](const GSL::Occupancy& occ)
0514:                                               {
0515:                                                   return occ == GSL::Occupancy::Free;
0516:                                               });
0517:         // This is a completely hacky arbitrary value to try and stop filaments from getting stuck right next to a wall
0518:         // ideally, we should implement a "deflection" instead so they move along the wall a bit rather than stopping dead
0519:         constexpr float wallStoppingProportion = 0.7;
0520:         currentPosition += movement * raycastInfo.distance * wallStoppingProportion;
0521: 
0522:         return true;
0523: #else
0524:         const auto& metadata = measuredHitProb.metadata;
0525:         bool pathIsFree = true;
0526: 
0527:         Vector2 vector = end - currentPosition;
0528:         float travelDistance = vmath::length(vector);
0529:         float stepSize = std::min(travelDistance, metadata.cellSize * 0.5f);
0530:         Vector2 increment = vmath::normalized(vector) * stepSize;
0531:         int steps = travelDistance / stepSize;
0532: 
0533:         int index = 0;
0534:         while (index < steps && pathIsFree)
0535:         {
0536:             currentPosition += increment;
0537:             index++;
0538:             Vector2Int pair = metadata.coordinatesToIndices(currentPosition.x, currentPosition.y);
0539:             pathIsFree = !metadata.indicesInBounds(pair) || measuredHitProb.freeAt(pair.x, pair.y);
0540:             if (!pathIsFree)
0541:                 currentPosition -= increment;
0542:         }
0543:         return pathIsFree;
0544: #endif
0545:     }

0596:     void Simulations::blurHitMap(cv::Mat& asImage) const
0597:     {
0598:         cv::GaussianBlur(asImage, asImage, cv::Size(0, 0), settings.blurSigmaX, settings.blurSigmaY);
0599:         // divide by the blurred mask to correct the edges always getting lower
0600:         // TODO measure performance of doing this repeatedly. Is it worth it to pre-compute and only redo it if the blur sigma has changed?
0601:         cv::Mat blurredMask;
0602:         cv::GaussianBlur(freeSpaceMask, blurredMask, cv::Size(0, 0), settings.blurSigmaX, settings.blurSigmaY);
0603: 
0604:         for (int i = 0; i < measuredHitProb.metadata.dimensions.y; i++)
0605:             for (int j = 0; j < measuredHitProb.metadata.dimensions.x; j++)
0606:                 if (blurredMask.at<float>(i, j) != 0)
0607:                     asImage.at<float>(i, j) = Utils::clamp(asImage.at<float>(i, j) / blurredMask.at<float>(i, j), 0, 1);
0608:     }
