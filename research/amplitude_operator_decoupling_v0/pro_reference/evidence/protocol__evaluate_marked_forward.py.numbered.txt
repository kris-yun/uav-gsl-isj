0019: def profiled_mark_ll(obs, pred):
0020:     obs=np.asarray(obs,float)
0021:     pred=np.asarray(pred,float)
0022:     m=obs>0
0023:     n=int(m.sum())
0024:     if n<4:
0025:         return 0.0
0026:     y=np.log(obs[m])
0027:     x=np.log(np.clip(pred[m],EPS,None))
0028:     xc=x-x.mean(); yc=y-y.mean()
0029:     den=float(xc@xc)
0030:     b=max(0.0,float(xc@yc/den)) if den>1e-12 else 0.0
0031:     a=float(y.mean()-b*x.mean())
0032:     r=y-(a+b*x)
0033:     rss=max(float(r@r),1e-12)
0034:     return -0.5*n*math.log(rss/n)
