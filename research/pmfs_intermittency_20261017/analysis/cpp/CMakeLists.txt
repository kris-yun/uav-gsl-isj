cmake_minimum_required(VERSION 3.16)
project(native_offline_replay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
add_executable(replay2026 replay2026.cpp native_functions.cpp)
add_executable(readout2026 readout2026.cpp native_functions.cpp)
add_executable(ordinary_controls ordinary_controls.cpp)
target_include_directories(replay2026 PRIVATE dda)
target_include_directories(readout2026 PRIVATE dda)
target_compile_options(replay2026 PRIVATE -O2)
target_compile_options(readout2026 PRIVATE -O2)
enable_testing()
add_test(NAME usage_exit COMMAND replay2026)
set_tests_properties(usage_exit PROPERTIES WILL_FAIL TRUE)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME historical_source_and_forward_parity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/validate_results.py)
